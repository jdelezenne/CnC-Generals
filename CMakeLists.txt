cmake_minimum_required(VERSION 3.25)
project(Generals LANGUAGES C CXX RC)

if(NOT WIN32 OR NOT MSVC OR NOT MSVC_VERSION EQUAL 1200)
    message(FATAL_ERROR "Use the original Windows Visual C++ 6.0 toolchain (_MSC_VER=1200). Modern MSVC needs a separate source port.")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "The original game and vendor ABIs require Win32/x86.")
endif()
if(NOT CMAKE_GENERATOR STREQUAL "NMake Makefiles")
    message(FATAL_ERROR "Use NMake: VC6 lacks the header dependency reporting required by Ninja.")
endif()
if(NOT CMAKE_BUILD_TYPE MATCHES "^(Debug|Release)$")
    message(FATAL_ERROR "Select Debug or Release with a VC6 preset.")
endif()
set(GEN_GAME "Generals" CACHE STRING "Game to build: Generals or ZeroHour")
set_property(CACHE GEN_GAME PROPERTY STRINGS Generals ZeroHour)
if(GEN_GAME STREQUAL "Generals")
    set(GEN_TREE Generals)
elseif(GEN_GAME STREQUAL "ZeroHour")
    set(GEN_TREE GeneralsMD)
else()
    message(FATAL_ERROR "GEN_GAME must be Generals or ZeroHour.")
endif()
set(GEN_CODE_DIR "${PROJECT_SOURCE_DIR}/${GEN_TREE}/Code")
option(GEN_CORE_ONLY "Build only the standalone Westwood libraries (no game executable)" OFF)
option(GEN_ENABLE_BINK "Enable RAD Bink movie playback" ON)
option(GEN_ENABLE_GAMESPY "Enable the original GameSpy online services" ON)
option(GEN_ENABLE_LZH "Enable the legacy LZH-Light compression format" ON)
option(GEN_ENABLE_BENCHMARK "Enable the original BYTEmark hardware benchmark" ON)
option(GEN_ENABLE_SAFEDISC "Enable Zero Hour's original SafeDisc SDK integration" OFF)

# The DSPs use the DLL CRT: /MD in Release and /MDd in Debug.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
set(CMAKE_MSVC_DEBUG_INFORMATION_FORMAT ProgramDatabase)
# CMake's VC6 Debug defaults add /GZ, which conflicts with the DSPs' optimized
# Debug Westwood libraries. Use each project's original optimization/check flags.
set(CMAKE_C_FLAGS_DEBUG "")
set(CMAKE_CXX_FLAGS_DEBUG "")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
include(cmake/Dependencies.cmake)
gen_find_dependencies()
if(NOT GEN_ENABLE_GAMESPY)
    set(GEN_GAMESPY_ROOT "${PROJECT_SOURCE_DIR}/cmake/stubs")
endif()
include(cmake/${GEN_TREE}Sources.cmake)
include(cmake/Targets.cmake)

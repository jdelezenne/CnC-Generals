cmake_minimum_required(VERSION 4.2)
project(Generals LANGUAGES C CXX RC)

if(NOT WIN32 OR NOT MSVC OR MSVC_VERSION LESS 1950)
    message(FATAL_ERROR "Use Visual Studio 2026 or newer on Windows.")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "The original game and vendor ABIs require Win32/x86.")
endif()
set(CMAKE_CONFIGURATION_TYPES "Debug;Release" CACHE STRING "Build configurations" FORCE)
set(GEN_GAME "Generals" CACHE STRING "Game to build: Generals or ZeroHour")
set_property(CACHE GEN_GAME PROPERTY STRINGS Generals ZeroHour)
if(GEN_GAME STREQUAL "Generals")
    set(GEN_TREE Generals)
elseif(GEN_GAME STREQUAL "ZeroHour")
    set(GEN_TREE GeneralsMD)
else()
    message(FATAL_ERROR "GEN_GAME must be Generals or ZeroHour.")
endif()
set(GEN_CODE_DIR "${PROJECT_SOURCE_DIR}/${GEN_TREE}/Code")
option(GEN_CORE_ONLY "Build only the standalone Westwood libraries (no game executable)" OFF)
option(GEN_ENABLE_BINK "Enable RAD Bink movie playback" ON)
option(GEN_ENABLE_GAMESPY "Enable the original GameSpy online services" OFF)
option(GEN_ENABLE_LZH "Enable the legacy LZH-Light compression format" OFF)
option(GEN_ENABLE_BENCHMARK "Enable the original BYTEmark hardware benchmark" OFF)
option(GEN_ENABLE_SAFEDISC "Enable Zero Hour's original SafeDisc SDK integration" OFF)

set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
set(CMAKE_MSVC_DEBUG_INFORMATION_FORMAT ProgramDatabase)
# Runtime checks are enabled per target to match the original Debug settings.
set(CMAKE_MSVC_RUNTIME_CHECKS "")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
foreach(config DEBUG RELEASE)
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
    set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/lib")
endforeach()
include(cmake/Dependencies.cmake)
gen_find_dependencies()
if(NOT GEN_ENABLE_GAMESPY)
    set(GEN_GAMESPY_ROOT "${PROJECT_SOURCE_DIR}/cmake/stubs")
endif()
include(cmake/${GEN_TREE}Sources.cmake)
include(cmake/STLport.cmake)
if(GEN_ENABLE_BINK AND NOT GEN_CORE_ONLY)
    add_subdirectory(vendors/libbinkdec)
endif()
include(cmake/Targets.cmake)

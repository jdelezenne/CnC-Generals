set(gpu_shader_dir "${CMAKE_BINARY_DIR}/Generated/GPU")
file(MAKE_DIRECTORY "${gpu_shader_dir}")
set(gpu_shaders)
foreach(shader FixedVertex FixedFragment PresentVertex PresentFragment)
    if(shader MATCHES "Vertex$")
        set(profile vs_6_0)
    else()
        set(profile ps_6_0)
    endif()
    set(source "${CMAKE_CURRENT_SOURCE_DIR}/Shaders/${shader}.hlsl")
    set(entry main)
    if(shader MATCHES "^Present")
        set(source "${CMAKE_CURRENT_SOURCE_DIR}/Shaders/Present.hlsl")
        string(REPLACE "Present" "" entry "${shader}")
        string(APPEND entry Main)
    endif()
    foreach(format Dxil Spirv)
        set(flags)
        if(format STREQUAL "Spirv")
            set(flags -spirv -fspv-target-env=vulkan1.0 -fvk-use-dx-layout)
        endif()
        set(binary "${gpu_shader_dir}/${shader}${format}.bin")
        set(header "${gpu_shader_dir}/${shader}${format}.h")
        add_custom_command(OUTPUT "${header}"
            COMMAND Vendor::DXC -T ${profile} -E ${entry} -O3 -Zpr ${flags} -Fo "${binary}" "${source}"
            COMMAND "${CMAKE_COMMAND}" "-DINPUT=${binary}" "-DOUTPUT=${header}" "-DSYMBOL=${shader}${format}" -P "${PROJECT_SOURCE_DIR}/CMake/EmbedShader.cmake"
            DEPENDS "${source}" "${PROJECT_SOURCE_DIR}/CMake/EmbedShader.cmake"
            VERBATIM)
        list(APPEND gpu_shaders "${header}")
    endforeach()
endforeach()
add_library(gen_shader_assembly STATIC LegacyAssembly.cpp)
target_link_libraries(gen_shader_assembly PRIVATE gen_graphics_headers)
target_compile_features(gen_shader_assembly PRIVATE cxx_std_23)
target_compile_definitions(gen_shader_assembly PRIVATE NOMINMAX)

add_library(gen_graphics STATIC D3D8.cpp LegacyShader.cpp ShaderCompiler.cpp ${gpu_shaders})
target_include_directories(gen_graphics PUBLIC "${PROJECT_SOURCE_DIR}" PRIVATE "${gpu_shader_dir}")
target_compile_features(gen_graphics PRIVATE cxx_std_23)
target_compile_definitions(gen_graphics PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
target_link_libraries(gen_graphics PUBLIC gen_platform gen_graphics_headers PRIVATE SDL3::SDL3-static Vendor::DXCompiler)

add_library(gen_d3dx STATIC D3DX8.cpp)
target_sources(gen_d3dx PRIVATE "${PROJECT_SOURCE_DIR}/Platform/SDL/ImageCodec.cpp")
target_compile_features(gen_d3dx PRIVATE cxx_std_23)
target_compile_definitions(gen_d3dx PRIVATE NOMINMAX)
target_link_libraries(gen_d3dx PUBLIC gen_graphics_headers PRIVATE gen_platform gen_shader_assembly DirectXTex SDL3_image::SDL3_image-static)
